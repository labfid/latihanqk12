Skrip Google Apps Script (GET & POST Data)

Buat Spreadsheet dengan 2 Sheet/Tab: Soal (Tab kedua) dan Jawaban (Tab pertama). Salin kode di bawah ke menu Extensions > Apps Script, lalu publikasikan sebagai Web App (Execute as: Me, Who has access: Anyone):

// 1. TARIK SOAL DARI SHEET "Soal" (GET Request)
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetSoal = ss.getSheetByName("Soal") || ss.getSheets()[1] || ss.getSheets()[0];
    var data = sheetSoal.getDataRange().getValues();
    
    var questions = [];
    for (var i = 1; i < data.length; i++) {
      if (!data[i][1]) continue;
      questions.push({
        Nomor: data[i][0] || i,
        Soal: data[i][1],
        OpsiA: data[i][2],
        OpsiB: data[i][3],
        OpsiC: data[i][4],
        OpsiD: data[i][5],
        OpsiE: data[i][6],
        KunciJawaban: data[i][7]
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      questions: questions
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

// 2. SIMPAN HASIL SANTRI KE SHEET "Jawaban" (POST Request)
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetJawaban = ss.getSheetByName("Jawaban") || ss.getSheets()[0];
    var data = JSON.parse(e.postData.contents);
    
    if (sheetJawaban.getLastRow() === 0) {
      sheetJawaban.appendRow(["Timestamp", "Nama Santri", "Kelas", "Skor", "Jumlah Pelanggaran", "Detail Jawaban"]);
    }
    
    sheetJawaban.appendRow([
      data.timestamp || new Date(),
      data.nama,
      data.kelas,
      data.skor,
      data.pelanggaran,
      JSON.stringify(data.detailJawaban)
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "status": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
